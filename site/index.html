How it works

Three lines of code to give your agent a cryptographic identity

import { AstraCipher } from '@astracipher/core';

const ap = new AstraCipher({ network: 'testnet' });

// Create a post-quantum secured agent identity
const { did, didId } = await ap.createAgent({
  name: 'my-trading-agent',
  description: 'Automated trading agent for equities',
});

console.log(didId);
// did:astracipher:testnet:a1b2c3d4e5f6...
// Issue a verifiable credential with capabilities
const credential = await ap.issueCredential({
  subjectDID: didId,
  capabilities: ['market-data:read', 'orders:execute'],
  permissions: [
    { resource: 'equity/*', actions: ['read', 'execute'] },
  ],
  trustLevel: 7,
  expiresIn: '90d',
});

// Credential is cryptographically signed with PQC keys
console.log(credential.proof.algorithm);
// "ML-DSA-65+ECDSA-P256"
// Verify an agent's credential  - works offline!
const result = await ap.verifyCredential(credential);

console.log(result.valid);       // true
console.log(result.checks);
// {
//   structure: { passed: true },
//   expiration: { passed: true },
//   signature: { passed: true },
//   revocation: { passed: true },
// }

// Check specific permissions
const allowed = ap.hasPermission(credential, 'equity/RELIANCE', 'execute');
// true  - wildcard equity/* matches
from astracipher import AstraCipherClient

async with AstraCipherClient(
    base_url="https://api.astracipher.com",
    api_key="ap_your_key",
) as client:
    # Create agent with PQC identity
    agent = await client.create_agent(name="my-agent")
    print(agent.did_id)
    # did:astracipher:testnet:a1b2c3d4...

    # Verify any credential
